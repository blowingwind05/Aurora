cmake_minimum_required(VERSION 3.16)

if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    project(AuroraEmbeddedPairingHost LANGUAGES CXX)
    include(CTest)
endif()
if(NOT DEFINED QT_VERSION_MAJOR)
    find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core Network)
endif()
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core Network)

if(NOT TARGET aurora_native_pairing)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/.." "${CMAKE_CURRENT_BINARY_DIR}/pairing")
endif()
if(NOT TARGET AuroraNativeTailscale)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../tailscale" "${CMAKE_CURRENT_BINARY_DIR}/tailscale")
endif()
if(NOT TARGET Aurora::SunshineLifecycle)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../sunshine/lifecycle" "${CMAKE_CURRENT_BINARY_DIR}/sunshine-lifecycle")
endif()

add_library(aurora_native_pairing_host STATIC EmbeddedPairingHost.cpp EmbeddedPairingHost.h)
add_library(Aurora::EmbeddedPairingHost ALIAS aurora_native_pairing_host)
set_target_properties(aurora_native_pairing_host PROPERTIES AUTOMOC ON CXX_EXTENSIONS OFF)
target_compile_features(aurora_native_pairing_host PUBLIC cxx_std_17)
target_include_directories(aurora_native_pairing_host PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(aurora_native_pairing_host PUBLIC aurora_native_pairing AuroraNativeTailscale Aurora::SunshineLifecycle)
if(MSVC)
    target_compile_options(aurora_native_pairing_host PRIVATE /utf-8)
endif()

if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR AND BUILD_TESTING)
    add_executable(aurora_pair_host_fixture tests/PairHostFixture.cpp)
    target_compile_features(aurora_pair_host_fixture PRIVATE cxx_std_17)
    target_link_libraries(aurora_pair_host_fixture PRIVATE Qt${QT_VERSION_MAJOR}::Core)
    add_executable(aurora_pair_host_test tests/PairHostTest.cpp)
    target_link_libraries(aurora_pair_host_test PRIVATE Aurora::EmbeddedPairingHost)
    add_test(NAME embedded_pairing_host_offline COMMAND aurora_pair_host_test
        $<TARGET_FILE:aurora_pair_host_fixture> "${CMAKE_CURRENT_SOURCE_DIR}/../../sunshine/admin/tests/tls")
    set_tests_properties(embedded_pairing_host_offline PROPERTIES TIMEOUT 30)
endif()
