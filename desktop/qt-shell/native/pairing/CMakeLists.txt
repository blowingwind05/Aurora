cmake_minimum_required(VERSION 3.16)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    project(AuroraNativePairing LANGUAGES CXX)
    include(CTest)
endif()

find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core Network)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core Network)
add_library(aurora_native_pairing STATIC PairingWire.cpp PairingWire.h Enrollment.cpp Enrollment.h)
target_compile_features(aurora_native_pairing PUBLIC cxx_std_17)
target_include_directories(aurora_native_pairing PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(aurora_native_pairing PUBLIC Qt${QT_VERSION_MAJOR}::Core Qt${QT_VERSION_MAJOR}::Network)

if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR AND BUILD_TESTING)
    add_executable(AuroraPairingWireTest PairingWireTest.cpp)
    target_link_libraries(AuroraPairingWireTest PRIVATE aurora_native_pairing)
    add_test(NAME pairing-wire-compatibility COMMAND AuroraPairingWireTest)
    add_executable(AuroraPairingEnrollmentTest EnrollmentTest.cpp)
    target_link_libraries(AuroraPairingEnrollmentTest PRIVATE aurora_native_pairing)
    add_test(NAME pairing-enrollment-compatibility COMMAND AuroraPairingEnrollmentTest)
endif()
